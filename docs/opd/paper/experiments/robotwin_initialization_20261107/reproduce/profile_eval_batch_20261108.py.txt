import json,time,statistics,os
from pathlib import Path
from types import SimpleNamespace
import torch
from toolkits.robotwin_mopd.worker_evaluate import compose_config
from rlinf.models.embodiment.openpi import get_model
from rlinf.envs.robotwin.multitask_env import RoboTwinMultiTaskEnv
S=Path(__file__).resolve().parent
out=S/'batch_profile_action20_20261108';out.mkdir(exist_ok=True)
tasks=['open_laptop','handover_mic','dump_bin_bigbin','stack_bowls_two']
runtime=Path(os.environ['ROBOTWIN_RUNTIME_ROOT'])
teachers={t:S/('candidate' if t=='open_laptop' else 'candidate_dump_bin_bigbin')/'torch'/t if t in ('open_laptop','dump_bin_bigbin') else runtime/'models/pi05_pace_torch'/t for t in tasks}
pool=json.loads((S/'shared_worker_eval_action20_20261107_r4/development50.json').read_text())
a=SimpleNamespace(output=out,override=[],checkpoint=teachers['stack_bowls_two'],norm_stats={t:str(p/'assets/robotwin/norm_stats.json') for t,p in teachers.items()},task='stack_bowls_two',gpu=1,episodes=1,pool=S/'shared_worker_eval_action20_20261107_r4/development50.json')
cfg=compose_config(a,pool)
torch.cuda.set_per_process_memory_fraction(.45)
model=get_model(cfg.rollout.model).to('cuda').eval()
env=RoboTwinMultiTaskEnv(cfg.env.eval,1,0,1,None)
entry=pool['tasks']['stack_bowls_two'][0]
env.set_rollout_tasks([dict(slot=0,task_id=0,task='stack_bowls_two',suite='stack_bowls_two',reset_seed=entry['scene_seed'],round=0)])
obs,_=env.reset(evaluation_episode=entry)
report={'device':torch.cuda.get_device_name(),'checkpoint':str(a.checkpoint),'note':'Repeated one real observation; model inference only, one resident simulator, not end-to-end parallel simulation. Full predict_action_batch with H25 / 10 flow ODE steps.','rows':[]}
for b in [1,2,4,8,16,32]:
 batch={k:v.repeat((b,)+(1,)*(v.ndim-1)) if torch.is_tensor(v) else v*b if isinstance(v,list) else v for k,v in obs.items()}
 try:
  with torch.no_grad():
   result=model.predict_action_batch(batch,mode='eval');del result
   torch.cuda.synchronize();torch.cuda.empty_cache();torch.cuda.reset_peak_memory_stats()
   times=[]
   for _ in range(3):
    torch.cuda.synchronize();t=time.perf_counter();result=model.predict_action_batch(batch,mode='eval');torch.cuda.synchronize();times.append(time.perf_counter()-t);del result
  row={'batch':b,'median_seconds':statistics.median(times),'seconds_per_observation':statistics.median(times)/b,'peak_allocated_gib':torch.cuda.max_memory_allocated()/2**30,'peak_reserved_gib':torch.cuda.max_memory_reserved()/2**30,'times':times}
 except torch.cuda.OutOfMemoryError:
  row={'batch':b,'oom_under_fraction':.45};torch.cuda.empty_cache()
 report['rows'].append(row);(out/'result.json').write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(row),flush=True)
 if 'oom_under_fraction' in row:break
env.close()
(out/'complete').write_text('OK\n')
